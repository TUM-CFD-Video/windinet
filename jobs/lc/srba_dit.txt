# if it is not adapter_q999, set `vae:` in configs/wan/srba_oooo/dit.yaml to it before (2).

sbatch --time=00:50:00 --job-name=lc_srba_smoke_wan jobs/lumi/lc/run.sbatch bash -c "torchrun --standalone --nproc_per_node=8 scripts/wan/train_dit.py configs/wan/srba_oooo/dit.yaml --name smoke/wan --desc 'smoke: 4 updates from Wan 2.1 on srba_oooo' --set train.steps=4 --set train.eval_every=2 --set train.log_every=1 --set data.eval_sims=16 --set data.test_sims=8 --set results_dir=/scratch/project_465003416/lc_work/smoke/srba_dit --set weights_dir=/scratch/project_465003416/lc_work/smoke_weights && torchrun --standalone --nproc_per_node=8 scripts/wan/train_dit.py configs/wan/dit_eval.yaml --set load=/scratch/project_465003416/lc_work/smoke/srba_dit/smoke/wan --set data.test_sims=8 --name smoke/wan/eval --desc 'smoke: dit_eval of smoke/wan'"
sbatch --time=00:50:00 --job-name=lc_srba_smoke_eu jobs/lumi/lc/run.sbatch bash -c "torchrun --standalone --nproc_per_node=8 scripts/wan/train_dit.py configs/wan/srba_oooo/dit.yaml --name smoke/eulermq --desc 'smoke: 4 unrolled updates from the eulermq DiT on srba_oooo' --set load=results/wan/eulermq/dit/res256/pairs/unrolled/unrolled2 --set train.unrolled=0.5 --set train.steps=4 --set train.eval_every=2 --set train.log_every=1 --set data.eval_sims=16 --set data.test_sims=8 --set results_dir=/scratch/project_465003416/lc_work/smoke/srba_dit --set weights_dir=/scratch/project_465003416/lc_work/smoke_weights && torchrun --standalone --nproc_per_node=8 scripts/wan/train_dit.py configs/wan/dit_eval.yaml --set load=/scratch/project_465003416/lc_work/smoke/srba_dit/smoke/eulermq --set data.test_sims=8 --name smoke/eulermq/eval --desc 'smoke: dit_eval of smoke/eulermq'"
# (3) first stage, 16000 updates at lr 1e-4 (warm-up 100, cosine), both starting points in parallel; about 17 h each.
sbatch --partition=standard-g --no-requeue --time=1-00:00:00 --job-name=lc_srba_dit_wan jobs/lumi/lc/run.sbatch torchrun --standalone --nproc_per_node=8 scripts/wan/train_dit.py configs/wan/srba_oooo/dit.yaml --name res256/pairs/recipe/wan16k --desc "Wan 2.1 1.3B fine-tuned on srba_oooo d4 latents: 16000 updates of 32 sims at lr 1e-4 (warm-up 100, cosine), conditioned on Mach, bubble count and shock direction" --set train.steps=16000 --set train.eval_every=1000
sbatch --partition=standard-g --no-requeue --time=1-00:00:00 --job-name=lc_srba_dit_eu jobs/lumi/lc/run.sbatch torchrun --standalone --nproc_per_node=8 scripts/wan/train_dit.py configs/wan/srba_oooo/dit.yaml --name res256/pairs/recipe/eulermq16k --desc "as wan16k, starting from the eulermq DiT res256/pairs/unrolled/unrolled2 (transformer weights; conditioning and AdamW fresh)" --set load=results/wan/eulermq/dit/res256/pairs/unrolled/unrolled2 --set train.steps=16000 --set train.eval_every=1000
# (4) unrolled stage from the final weights of the chosen <run> (keeps its AdamW moments); about 6 h.
sbatch --partition=standard-g --no-requeue --time=10:00:00 --job-name=lc_srba_unrolled jobs/lumi/lc/run.sbatch torchrun --standalone --nproc_per_node=8 scripts/wan/train_dit.py configs/wan/dit_anneal.yaml --set load=results/wan/srba_oooo/dit/res256/pairs/recipe/<run> --set train.steps=4000 --set train.lr=1e-5 --set train.patchify_lr=1e-5 --set train.warmup_steps=100 --set train.unrolled=0.5 --set train.seed=3 --name res256/pairs/recipe/<run>/unrolled --desc "from <run>, 4000 updates at 1e-5 keeping its AdamW moments: half of each micro-batch trained on its own sampler state (unrolled 0.5)"
# (5) test evaluation, once, of the run chosen on val: all 60 test sims, 5 steps, eta 1; about 20 min.
sbatch --partition=standard-g --no-requeue --time=01:00:00 --job-name=lc_srba_bench jobs/lumi/lc/run.sbatch torchrun --standalone --nproc_per_node=8 scripts/wan/train_dit.py configs/wan/dit_eval.yaml --set load=results/wan/srba_oooo/dit/res256/pairs/recipe/<run>/unrolled --name res256/pairs/recipe/<run>/unrolled/bench_steps5_eta1 --desc "test eval of <run>/unrolled: 5 steps, eta 1"
