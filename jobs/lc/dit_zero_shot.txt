# 256 = our simulation format (Wan's native 832x480 skipped for speed); abstract = a prompt without physics;
# gt = test sim 0250 (gamma 1.4) as the rgb3 adapter image through the frozen VAE; sdedit_x = the same noised to x and denoised.
python scripts/wan/sample_dit.py configs/wan/sample.yaml --name zs_256 --desc "zero-shot, physics prompt, 256x256x101 (our format)" --set height=256 --set width=256 --set frames=101
python scripts/wan/sample_dit.py configs/wan/sample.yaml --name zs_256_abstract --desc "zero-shot, abstract prompt, 256x256x101" --set height=256 --set width=256 --set frames=101 --set prompt="A colourful abstract animation of slowly flowing and swirling shapes."
python scripts/wan/sample_dit.py configs/wan/sample.yaml --name gt_0250 --desc "ground truth: test sim 0250 (gamma 1.4) as rgb3 adapter image, frozen VAE round trip" --set source=euler_mq_dataset/256x256_ds/test.h5:0250_gamma1.3999999762 --set adapter=results/wan/vae/res256/rgb3/adapter --set strength=0
python scripts/wan/sample_dit.py configs/wan/sample.yaml --name sdedit_0250_0.5 --desc "test sim 0250 noised to strength 0.5 and denoised by the pretrained DiT with the physics prompt" --set source=euler_mq_dataset/256x256_ds/test.h5:0250_gamma1.3999999762 --set adapter=results/wan/vae/res256/rgb3/adapter --set strength=0.5
python scripts/wan/sample_dit.py configs/wan/sample.yaml --name sdedit_0250_0.8 --desc "test sim 0250 noised to strength 0.8 and denoised by the pretrained DiT with the physics prompt" --set source=euler_mq_dataset/256x256_ds/test.h5:0250_gamma1.3999999762 --set adapter=results/wan/vae/res256/rgb3/adapter --set strength=0.8
python scripts/wan/sample_dit.py configs/wan/sample.yaml --name firstframe_0250 --desc "test sim 0250: first latent frame kept clean at timestep 0, the rest generated from noise by the pretrained DiT (the fine-tune's conditioning, zero-shot)" --set source=euler_mq_dataset/256x256_ds/test.h5:0250_gamma1.3999999762 --set adapter=results/wan/vae/res256/rgb3/adapter --set keep_first_frame=true
